cmake_minimum_required(VERSION 3.22)

project(asio-ice)

option(ICE_TEST "Build test" ON)

set(CMAKE_CXX_STANDARD 23)
set(STANDALONE_ASIO_DIR "E:\\openSource\\asio")
# set(STDEXEC_DIR "E:\\openSource\\asio2exec\\stdexec")
# set(Boost_DIR "D:\\openSource\\boost_1_88_0_prebuild\\lib64-msvc-14.3\\cmake\\Boost-1.88.0")
set(ASIOICE_USE_BOOST_ASIO ON)
# set(OPENSSL_DIR "E:\\openSource\\openssl-3.1.0\\x64")
# set(OPENSSL_LIB_DIR ${OPENSSL_DIR}/lib)
# set(OPENSSL_INCLUDE_DIR ${OPENSSL_DIR}/include)
set(ENABLE_IO_URING OFF)

find_library(
    OPENSSL_SSL_LIB 
    NAMES ssl
    PATHS ${OPENSSL_LIB_DIR}
)
find_library(
    OPENSSL_CRYPTO_LIB 
    NAMES crypto
    PATHS ${OPENSSL_LIB_DIR}
)
if(NOT OPENSSL_SSL_LIB OR NOT OPENSSL_CRYPTO_LIB)
    message(FATAL_ERROR "Could not find OpenSSL libraries in ${OPENSSL_LIB_DIR}")
else()
    message(STATUS "Found OpenSSL.")
endif()

if(ASIOICE_USE_BOOST_ASIO)
    set(__ASIOICE_USE_BOOST_ASIO 1)
    # set(CPPMDNS_USE_BOOST_ASIO ON)
else()
    set(__ASIOICE_USE_BOOST_ASIO 0)
    # set(CPPMDNS_USE_BOOST_ASIO OFF)
endif()

set(Boost_DEBUG ON)
set(Boost_USE_STATIC_LIBS ON)
find_package(Boost 1.89 CONFIG REQUIRED COMPONENTS json context)
if(Boost_FOUND)
    message(STATUS "Boost found")
endif()

if(ENABLE_IO_URING)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        set(ASIOICE_ENABLE_IO_URING 1)
        message(STATUS "liburing support enabled")
    else()
        set(ASIOICE_ENABLE_IO_URING 0)
        message(WARNING "ENABLE_IO_URING is ON, but system is not Linux. Disabling.")
    endif()
else()
    set(ASIOICE_ENABLE_IO_URING 0)
endif()
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/include/config.hpp.in" "${CMAKE_CURRENT_SOURCE_DIR}/include/config.hpp")

# add_subdirectory(cppmdns)

file(GLOB ICE_SRC_FILES 
    src/stun.cpp
    src/base64.cpp
    src/address.cpp
    src/candidate.cpp
    src/hash.cpp
    src/candidate_pair.cpp
    src/boost_context/context.cpp
    src/ssl/datagram_bio.cpp
    src/string_utils.cpp
    src/io_buffer2.cpp
)
add_library(asioice STATIC ${ICE_SRC_FILES})

if(ASIOICE_ENABLE_IO_URING)
    target_link_libraries(asioice PUBLIC uring)
endif()

add_library(asioice_compile_flags INTERFACE)
target_compile_options(asioice_compile_flags INTERFACE
                       $<$<COMPILE_LANG_AND_ID:CXX,GNU>:-fconcepts-diagnostics-depth=10 -Wno-non-template-friend -Wall -fcoroutines>
                       )
target_compile_options(asioice_compile_flags INTERFACE
                       $<$<COMPILE_LANG_AND_ID:CXX,MSVC>:/Zc:__cplusplus /Zc:preprocessor /D_WIN32_WINNT=0x0601 /Zm200 /bigobj /W4 /wd4100 /wd4101 /wd4127 /wd4324 /wd4456 /wd4459>
                       )

target_link_libraries(asioice PRIVATE ${OPENSSL_SSL_LIB} ${OPENSSL_CRYPTO_LIB} Boost::boost Boost::context asioice_compile_flags)
if(NOT ASIOICE_USE_BOOST_ASIO)
    target_include_directories(asioice PUBLIC ${STANDALONE_ASIO_DIR}/asio/include)
endif()
target_include_directories(asioice PUBLIC ${STDEXEC_DIR})
target_include_directories(asioice PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_include_directories(asioice PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_include_directories(asioice PRIVATE ${OPENSSL_INCLUDE_DIR})
target_include_directories(asioice PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party)

add_library(test_flags INTERFACE)
target_link_libraries(test_flags INTERFACE ${OPENSSL_SSL_LIB} ${OPENSSL_CRYPTO_LIB} Boost::boost Boost::context)
if(NOT ASIOICE_USE_BOOST_ASIO)
    target_include_directories(test_flags INTERFACE ${STANDALONE_ASIO_DIR}/asio/include)
endif()
target_link_libraries(test_flags INTERFACE asioice asioice_compile_flags)
target_include_directories(test_flags INTERFACE ${STDEXEC_DIR})
target_include_directories(test_flags INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_include_directories(test_flags INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_include_directories(test_flags INTERFACE ${OPENSSL_INCLUDE_DIR})
target_include_directories(test_flags INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/third_party)

file(GLOB TEST_SOURCES "src/*_test.cpp")
if(ICE_TEST)
    foreach(TEST_FILE ${TEST_SOURCES})
        get_filename_component(EXECUTABLE_NAME ${TEST_FILE} NAME_WE)
        add_executable(${EXECUTABLE_NAME} ${TEST_FILE})
        target_link_libraries(${EXECUTABLE_NAME} PRIVATE test_flags)
    endforeach()
endif()
